  params <- base::list("query" = query, "search_type" = search_type, "similarity_type" = similarity_type, "min_similarity" = min_similarity, "hazard_name" = hazard_name, "min_toxicity" = min_toxicity, "min_authority" = min_authority, "mass_type" = mass_type, "min_mass" = min_mass, "max_mass" = max_mass, "filter_features" = filter_features, "feature_filters" = feature_filters, "element_include" = element_include, "element_exclude" = element_exclude, "exclude_all_others" = exclude_all_others, "limit" = limit, "all_pages" = all_pages)

  changed <- base::intersect(base::names(params), base::names(state$params))
  params[changed] <- state$params[changed]

  state["result"] <- base::list(result)
  result <- run_hook("chemi_search", "post_response", state)
  result
}
